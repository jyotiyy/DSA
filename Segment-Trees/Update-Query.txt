updateSgmentTree(idx,value,i,l,r){
    if(l == r){
        if(l == idx){
            segTree[i] = value;
            
        }
        return

    }
    mid = (l+r)/2
    if(idx <= mid){ //left child
        updateSegTree(idx,value,2*i+1,l,m)
    }else{
        updateSegTree(idx,value,2*i+2,m+1,r)
    }
    segTree[i] = segTree[2*i+1]+segTree[2*i+2]

}
